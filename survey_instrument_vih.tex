\section*{A. Introduction and Instructions}

\textbf{Welcome to this survey about COVID-19 vaccination preferences.}

This survey asks you to make choices between different hypothetical COVID-19 vaccine options. There are no right or wrong answers---we are interested in understanding your preferences.

\textbf{How to complete this survey:}
\begin{itemize}
    \item You will see \textbf{6 choice tasks}
    \item Each task presents \textbf{two vaccine alternatives (A and B)} and an option to \textbf{choose neither}
    \item Each vaccine is described by 5 attributes: waiting time, effectiveness, side effects, origin, and cash incentive
    \item Please select the option you would prefer if these were your real choices
\end{itemize}

\textbf{Your responses are confidential and will only be used for research purposes.}

\vspace{1em}
\noindent\textbf{Estimated time:} 10--15 minutes

\section*{C. Attribute Definitions}

\begin{description}
    \item[\textbf{Waiting time to vaccination}] The time between your decision to get vaccinated and the actual vaccination appointment.
    \begin{itemize}
        \item 0 months = Walk-in appointment (no wait)
        \item 1, 2, 3, or 6 months = Scheduled appointment after waiting period
    \end{itemize}
    
    \item[\textbf{Vaccine efficacy}] The vaccine's effectiveness in preventing COVID-19 infection (based on clinical trial data).
    \begin{itemize}
        \item 50\%, 70\%, or 95\% protection rate
    \end{itemize}
    
    \item[\textbf{Side-effect profile}] Expected side effects after vaccination.
    \begin{itemize}
        \item \textbf{Mild:} Fatigue lasting 1--2 days
        \item \textbf{Moderate:} Fever lasting approximately 2 days
        \item \textbf{Severe:} Fatigue lasting approximately 1 week
    \end{itemize}
    
    \item[\textbf{Vaccine origin}] Country of manufacture.
    \begin{itemize}
        \item Domestic = Manufactured in China
        \item Imported = Manufactured abroad
    \end{itemize}
    
    \item[\textbf{Cash incentive}] One-time payment received after completing vaccination.
    \begin{itemize}
        \item 50, 200, or 800 RMB (Chinese Yuan)
    \end{itemize}
\end{description}
